# 今日宜素材图标准

#!/usr/bin/env bash
# pre-commit: 工作台写入约束校验  v2.1
# 规程依据: public/consensus/rules.md 第八条
#
# v1 问题: git diff --cached --name-only 对中文/空格/特殊字符文件名会
#   加双引号包 + 八进制转义 (\350\220...)，read 拿到的字符串 startswith
#   全部不匹配，silent pass（已复现：549d590 本应被拦却放行）。
# v2 修: git diff --cached -z (NUL 分隔)，避免一切包装；解八进制兜底；
#   加"暂存区改动但解析为零"的检查防 silent pass。
# v2.1 修: macOS /bin/bash 的 set -u 把中文句号"。"视为变量名一部分，
#   echo "维护人: $m。他人..." 会查找 ${m。} 然后 unbound variable。
#   全部改用 printf 显式格式 + ${var} 大括号包住所有引用。
set -u

# ── 小工具：bash 下解八进制转义 ────────────────────────────
_decode_octal_path() {
  local s="$1"
  if [[ "$s" != *\\* ]]; then
    printf '%s' "$s"
    return
  fi
  # shellcheck disable=SC2059
  printf '%b' "$s"
}

# 1. 识别当前提交者身份
author_name=$(git config user.name)
case "${author_name}" in
  alice) me="alice" ;;
  bob)   me="bob"   ;;
  carol) me="carol" ;;
  dave)  me="dave"  ;;
  *)
    printf 'ERROR: 未知身份 "%s"。请用 alice/bob/carol/dave 之一。\n' "${author_name}"
    printf '       在克隆内执行: git config user.name <name> && git config user.email <email>\n'
    exit 1
    ;;
esac

# 2. 公共区文件维护人映射
maintainer_of() {
  case "$1" in
    public/consensus/project-scope.md) printf 'alice' ;;
    public/consensus/rules.md)         printf 'alice' ;;
    public/consensus/glossary.md)      printf 'alice' ;;
    public/contract/division.md)       printf 'bob'   ;;
    public/contract/protocol.md)       printf 'carol' ;;
    *) printf '' ;;
  esac
}

errors=0
parsed_count=0

# 3. 遍历暂存区新增/修改/重命名的文件（-z NUL 分隔，无引号无转义包装）
while IFS= read -r -d '' raw; do
  raw="${raw#\"}"
  raw="${raw%\"}"
  file=$(_decode_octal_path "$raw")
  parsed_count=$((parsed_count+1))

  # 3.1 成员区写入权限校验
  if [[ "$file" == members/* ]]; then
    member=$(printf '%s' "$file" | cut -d'/' -f2)
    rest=$(printf '%s'   "$file" | cut -d'/' -f3-)

    if [[ "$member" == "${me}" ]]; then
      : # 本人成员区，放行
    elif [[ "$rest" == inbox/* ]]; then
      : # 他人收件箱，允许投递
    else
      printf 'ERROR: %s 不可改 members/%s/ 下非 inbox 文件:\n' "${me}" "${member}"
      printf '       %s\n' "${file}"
      printf '       规程: 跨区协作只投对方 inbox，不改对方区已有文件。\n'
      errors=$((errors+1))
    fi
    continue
  fi

  # 3.2 公共区维护人校验
  if [[ "$file" == public/consensus/* ]] || [[ "$file" == public/contract/* ]]; then
    m=$(maintainer_of "$file")
    if [[ -n "${m}" ]] && [[ "${m}" != "${me}" ]]; then
      printf 'ERROR: %s 不可改 %s\n' "${me}" "${file}"
      printf '       维护人: %s — 他人修改须向维护人提出，由维护人确认后更新。\n' "${m}"
      errors=$((errors+1))
    fi
    continue
  fi

  # 3.3 其他路径（.githooks/ 等）放行
done < <(git diff --cached -z --name-only --diff-filter=ACMR)

# 3.4 防御：暂存区有改动但解析为 0 → 解析失败，拒绝放行
staged_count=$(git diff --cached --name-only --diff-filter=ACMR 2>/dev/null | wc -l | tr -d ' ')
if [[ -n "${staged_count:-}" ]] && [[ "${staged_count}" -gt 0 ]] && [[ "${parsed_count}" -eq 0 ]]; then
  printf 'ERROR: pre-commit 无法解析暂存区内 %s 个文件的路径（可能是 git 版本差异/编码问题）。\n' "${staged_count}"
  printf '       为避免越权写入，拒绝放行。请提交 plain ASCII 文件名重试，或联系工作台维护人检查 hook。\n'
  errors=$((errors+1))
fi

# 4. 结果
if [[ "${errors}" -gt 0 ]]; then
  printf '\n'
  printf 'pre-commit 拦截 %s 处违规。依据 public/consensus/rules.md，请调整后重试。\n' "${errors}"
  printf '若需绕过（仅搭建期），用: git commit --no-verify （不建议常规使用）\n'
  exit 1
fi

exit 0

1. 公共区一块：放项目共识文件与协作契约。
2. 成员区人手一块：区内文件只有本人写入。
3. 公共区项目共识文件每份只有一个维护人，别人要改须向维护人提出。

1. 跨区协作只有一种动作：把要办的事写成一份新文件，投进对方收件箱。

3. 对方办结后归档，再投一份答复到发起人收件箱。

## 四、生效路径
1. 人指定任务目标与合格标准。
2. AI 在成员区内起草（草稿反复修改，不对外生效）。

4. 生效版本全空间唯一。替换前先把旧版原样存入历史版本，历史版本只增不删。

1. 同一信息只在一个位置维护：当前进展看状态记录，正式口径看生效版本，过程看当天日志，其余位置只作引用。
2. 一项工作是否生效，只看有没有经人确认的生效版本；文件夹与草稿的存在说明不了任何事。

# 协作规则

> 协作契约：全员共同遵守。
> 维护人：carol（服务端&后台开发）。

## 一、跨区投递格式

投递件文件名：`YYYY-MM-DD-<发起人>-<事由>.md`，例如 `2026-08-31-alice-今日宜素材图标准确认.md`

投递件须含三要素：
```markdown
# <事由>

## 需要确认什么
<列出需要对方拍板的具体问题>

## 建议口径
<给出建议的处理方案>

## 确认后写回哪里
<写明对方确认后，答复应投回发起人收件箱的文件名>
```

## 二、生效路径标准动作

1. 起草：AI 在发起人 drafts/ 下起草，文件名 `<主题>-draft-vN.md`
2. 提请确认：发起人把"提请确认"投进对方 inbox
3. 对方确认：
   - 通过 → 对方在自己的 effective/ 下不直接写（生效版本归发起人本人维护）；对方在 inbox 答复"通过"
   - 打回 → 对方在 inbox 答复"打回 + 修改意见"
4. 发起人生效：
   - 把 drafts/ 草稿复制到 effective/（同名去 -draft 后缀）
   - 若 effective/ 已有旧版，先把旧版原样复制到 history/<主题>/<日期>_vN.md
   - 在 status.md 记录生效
5. 归档：草稿可保留在 drafts/，或移入 history/（团队约定：保留在 drafts/，便于追溯）

## 三、状态记录格式（status.md）

```markdown
# <成员> 状态记录

## 当前进展
- YYYY-MM-DD <事项> <状态：进行中/待确认/已生效/已归档>

## 生效版本清单
- <文件名> | 生效日期 | 版本
```

## 四、当天日志格式（log/YYYY-MM-DD.md）

```markdown
# <成员> YYYY-MM-DD 日志

## 今日工作
- <事项> <动作> <结果>

## 提交记录
- <commit hash 前 7 位> <commit message>
```

## 五、确认与签字

人确认生效版本时，须在 status.md 写明：
```
- YYYY-MM-DD <文件名> 经 <确认人> 确认通过，生效
```
确认人签名即生效，无须额外流程。
